class SegTree:
    def __init__(self,function,initial_value,arraysize):
        self.op = function
        self.datasize = 1
        while self.datasize <= arraysize:
            self.datasize *= 2
        self.data = [initial_value] * (self.datasize*2)

    def set(self,nth,value):
        self.idx = self.datasize + nth

#===============================================================================
#       値をsetするときに、値をvalueに新しく更新するか、値にvalueを加算するか、などはここで
#       調整すること。

        self.data[self.idx] = self.op(self.data[self.idx], value)
#        self.data[self.idx] = value
#===============================================================================

        while self.idx > 1:
            if self.idx % 2 == 0:
                self.data[self.idx//2] = self.op(self.data[self.idx],self.data[self.idx+1])
            else:
                self.data[self.idx//2] = self.op(self.data[self.idx],self.data[self.idx-1])
            self.idx //= 2
        return None

    def prod(self,l,r):
        if l > r:
            raise ValueError("l=",l,"r=",r,"l>rの値が指定されており範囲指定が不正です。arrayサイズが1のセグメント木を構築していませんか")
        self.left = l + self.datasize
        self.right = r + self.datasize

        if self.left == self.right:
            return self.data[self.left]

        self.winleft = self.data[self.left]
        self.winright = self.data[self.right]

        while self.left//2 != self.right//2:
            if self.left%2 == 0:
                if self.left+1 <= self.right:
                    self.winleft = self.op(self.winleft,self.data[self.left+1])
            self.left //= 2

            if self.right % 2 == 1:
                if self.right - 1 >= self.left:
                    self.winright = self.op(self.winright,self.data[self.right-1])
            self.right //= 2

        return self.op(self.winleft,self.winright)

#=====初期化で渡す関数========

def segmax(seg_a,seg_b):
    return max(seg_a,seg_b)

def segmin(seg_a,seg_b):
    return min(seg_a,seg_b)

def segsum(seg_a,seg_b):
    return seg_a + seg_b
